### 4.2 星级评分
```csharp
public int CalculateStar(LevelData level, PlayerResult result)
{
    int star = 0;
    
    // 1星: 完成关卡目标
    if (result.IsCompleted)
        star = 1;
    
    // 2星: 达成目标 + 使用步数<限制*1.5
    if (star >= 1 && result.UsedMoves <= level.MaxMoves * 1.5f)
        star = 2;
    
    // 3星: 达成目标 + 使用步数<限制*1.2
    if (star >= 2 && result.UsedMoves <= level.MaxMoves * 1.2f)
        star = 3;
    
    return star;
}
```

### 8.2 签到逻辑
```csharp
public class SignInSystem
{
    private int m_currentDay;      // 连续签到天数
    private long m_lastSignTime;  // 上次签到时间
    private bool[] m_rewardClaimed = new bool[7];
    
    public bool CanSignIn()
    {
        // 检查是否已签到
        if (IsTodaySigned())
            return false;
            
        // 检查断签(超过48小时断签)
        if (IsMoreThan48Hours())
            m_currentDay = 0;
            
        return true;
    }
    
    public void SignIn()
    {
        m_currentDay++;
        m_lastSignTime = TimeUtil.GetCurrentTime();
        m_rewardClaimed[m_currentDay - 1] = true;
    }
}
```

### 8.3 补签机制
- 补签卡道具: 使用1张补签卡可补签1天
- 补签价格: 10钻石/次
- 断签重置: 超过48小时未签到，连续天数重置
